test attributes
================================================================================

@test
let () -> adds -> () {
    assert (add(-1, -2) = -3);
}

@test(traps <- "x", timeout <- 2)
let () -> overflows -> () {
    _ <- square(4294967296);
}

let () -> outer -> () {
    @test
    let () -> inner -> () { assert (square(3) = 9); }
}
